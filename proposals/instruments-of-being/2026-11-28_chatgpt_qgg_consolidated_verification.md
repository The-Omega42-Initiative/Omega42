import numpy as np
from itertools import product
pts=list(product(range(3),repeat=3)); N=81

R=np.array([[0,-1,0],[1,0,0],[0,0,1]])
X=np.array([[1,0,0],[0,0,-1],[0,1,0]])
for i,U in enumerate([R,X,np.eye(3)]):
    s=np.zeros((N,N))
    for a,p in enumerate(pts):
        q=list(p); q[i]=(q[i]+1)%3; b=pts.index(tuple(q))
        s[3*a:3*a+3,3*b:3*b+3]=U
    S.append(s)
F=[s-I for s in S]; D=[(s-s.T)/2 for s in S]
A=[(I+s.T)/2 for s in S]; L=[f.T@f for f in F]
comm=lambda a,b:a@b-b@a
for i,j in [(0,1),(0,2),(1,2)]:
    K=comm(F[i],F[j])
    C=comm(D[i],D[j])
    rhs=A[i]@A[j]@K+A[i]@comm(F[i],A[j])@F[j]-A[j]@comm(F[j],A[i])@F[i]+comm(A[i],A[j])@F[j]@F[i]

    assert np.allclose(4*np.linalg.norm(C)**2,np.linalg.norm(K)**2)

    assert np.allclose(u.conj().T@u,np.eye(2))
